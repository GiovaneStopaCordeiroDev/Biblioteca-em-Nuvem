  <script>
    let listaLivros = [];
    let mockData = [
      { id: 1, titulo: "O Cortiço", autor: "Aluísio Azevedo", isbn: "978-8508041534", genero: "Romance", ano: 1890, quantidade: 3 },
      { id: 2, titulo: "Memórias Póstumas de Brás Cubas", autor: "Machado de Assis", isbn: "978-8535910663", genero: "Ficção", ano: 1881, quantidade: 5 }
    ];

    document.addEventListener("DOMContentLoaded", () => {
      carregarLivros();
    });

    function showAlert(message, type = 'success') {
      const alertBox = document.getElementById('global-alert');
      alertBox.className = `alert alert-${type}`;
      alertBox.style.display = 'flex';
      alertBox.innerHTML = `<i class="fa-solid fa-${type === 'success' ? 'circle-check' : 'triangle-exclamation'}"></i> ${message}`;
      
      setTimeout(() => { alertBox.style.display = 'none'; }, 4000);
    }

    function getHeaders() {
      const token = document.getElementById('api-token').value.trim();
      const headers = { 'Content-Type': 'application/json' };
      if (token) {
        headers['Authorization'] = token.startsWith('Bearer ') ? token : `Bearer ${token}`;
      }
      return headers;
    }

    function getApiUrl() {
      return document.getElementById('api-url').value.trim();
    }

    function isMock() {
      return document.getElementById('mock-mode').checked;
    }

    // 1. LISTAR LIVROS
    async function carregarLivros() {
      const tbody = document.getElementById('books-tbody');
      tbody.innerHTML = `<tr><td colspan="5" style="text-align:center; padding: 20px;"><i class="fa-solid fa-spinner spinner"></i> Carregando acervo...</td></tr>`;

      if (isMock()) {
        listaLivros = [...mockData];
        renderizarTabela(listaLivros);
        return;
      }

      try {
        const response = await fetch(getApiUrl(), { headers: getHeaders() });
        if (!response.ok) throw new Error(`Status ${response.status}`);
        
        const data = await response.json();
        
        listaLivros = data.map(item => ({
          id: item.id || item._id,
          titulo: item.titulo || item.title || "Sem título",
          autor: item.autor || item.author || "Desconhecido",
          isbn: item.isbn || "N/A",
          genero: item.genero || item.genre || "Geral",
          ano: item.ano || item.year || item.ano_publicacao || "N/A",
          quantidade: item.quantidade !== undefined ? item.quantidade : (item.stock || 1)
        }));

        renderizarTabela(listaLivros);
      } catch (err) {
        console.error("Erro ao buscar livros:", err);
        tbody.innerHTML = `<tr><td colspan="5" style="text-align:center; color: var(--danger); padding: 20px;">Não foi possível conectar ao servidor. Verifique a URL ou selecione o modo Mock.</td></tr>`;
      }
    }

    function renderizarTabela(livros) {
      const tbody = document.getElementById('books-tbody');
      tbody.innerHTML = '';

      if (livros.length === 0) {
        tbody.innerHTML = `<tr><td colspan="5" style="text-align:center; color: var(--cinza); padding: 20px;">Nenhum livro encontrado.</td></tr>`;
        return;
      }

      livros.forEach(livro => {
        tbody.innerHTML += `
          <tr>
            <td>
              <strong>${livro.titulo}</strong>
              <div style="font-size: 12px; color: var(--cinza);">${livro.autor} • <span class="badge">${livro.genero}</span></div>
            </td>
            <td><code>${livro.isbn}</code></td>
            <td>${livro.ano}</td>
            <td><strong>${livro.quantidade}</strong></td>
            <td style="text-align: right;">
              <button class="btn btn-editar btn-sm" onclick="prepararEdicao(${livro.id})">
                <i class="fa-solid fa-pen-to-square"></i> Editar
              </button>
              <button class="btn btn-danger btn-sm" onclick="deletarLivro(${livro.id})">
                <i class="fa-solid fa-trash"></i>
              </button>
            </td>
          </tr>
        `;
      });
    }

    // 2. SALVAR (CRIAR / EDITAR)
    async function salvarLivro(event) {
      event.preventDefault();

      const id = document.getElementById('book-id').value;
      const livroPayload = {
        titulo: document.getElementById('titulo').value.trim(),
        autor: document.getElementById('autor').value.trim(),
        isbn: document.getElementById('isbn').value.trim(),
        genero: document.getElementById('genero').value.trim() || "Geral",
        ano: parseInt(document.getElementById('ano').value),
        quantidade: parseInt(document.getElementById('quantidade').value)
      };

      const isEdit = Boolean(id);
      const url = isEdit ? `${getApiUrl()}/${id}` : getApiUrl();
      const method = isEdit ? 'PUT' : 'POST';

      const btnSave = document.getElementById('btn-save');
      btnSave.disabled = true;
      btnSave.innerHTML = `<i class="fa-solid fa-spinner spinner"></i> Salvando...`;

      if (isMock()) {
        setTimeout(() => {
          if (isEdit) {
            const idx = mockData.findIndex(l => l.id == id);
            if (idx !== -1) mockData[idx] = { id: Number(id), ...livroPayload };
            showAlert("Livro atualizado com sucesso!");
          } else {
            const newId = mockData.length ? Math.max(...mockData.map(l => l.id)) + 1 : 1;
            mockData.push({ id: newId, ...livroPayload });
            showAlert("Livro adicionado com sucesso!");
          }
          finalizarSalvar();
        }, 400);
        return;
      }

      try {
        const response = await fetch(url, {
          method: method,
          headers: getHeaders(),
          body: JSON.stringify(livroPayload)
        });

        if (!response.ok) throw new Error("Erro ao salvar no servidor.");

        showAlert(isEdit ? "Livro atualizado com sucesso!" : "Livro adicionado com sucesso!");
        finalizarSalvar();
      } catch (err) {
        console.error(err);
        showAlert("Falha ao salvar no servidor backend.", "error");
        btnSave.disabled = false;
        btnSave.innerHTML = `<i class="fa-solid fa-plus"></i> ${isEdit ? 'Salvar Alterações' : 'Adicionar Livro'}`;
      }
    }

    function finalizarSalvar() {
      const btnSave = document.getElementById('btn-save');
      btnSave.disabled = false;
      limparFormulario();
      carregarLivros();
    }

    function prepararEdicao(id) {
      const livro = listaLivros.find(l => l.id == id);
      if (!livro) return;

      document.getElementById('book-id').value = livro.id;
      document.getElementById('titulo').value = livro.titulo;
      document.getElementById('autor').value = livro.autor;
      document.getElementById('isbn').value = livro.isbn;
      document.getElementById('genero').value = livro.genero;
      document.getElementById('ano').value = livro.ano;
      document.getElementById('quantidade').value = livro.quantidade;

      document.getElementById('form-title').innerHTML = `<i class="fa-solid fa-pen-to-square" style="color: var(--verde-principal);"></i> Editar Livro`;
      
      const btnSave = document.getElementById('btn-save');
      btnSave.innerHTML = `<i class="fa-solid fa-floppy-disk"></i> Salvar Alterações`;
      
      document.getElementById('btn-cancel').style.display = 'inline-block';
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    // 3. DELETAR
    async function deletarLivro(id) {
      if (!confirm("Tem certeza que deseja excluir este livro?")) return;

      if (isMock()) {
        mockData = mockData.filter(l => l.id != id);
        showAlert("Livro removido do acervo local!");
        carregarLivros();
        return;
      }

      try {
        const response = await fetch(`${getApiUrl()}/${id}`, {
          method: 'DELETE',
          headers: getHeaders()
        });

        if (!response.ok) throw new Error("Erro ao deletar.");

        showAlert("Livro removido com sucesso!");
        carregarLivros();
      } catch (err) {
        console.error(err);
        showAlert("Erro ao excluir o livro do servidor.", "error");
      }
    }

    function limparFormulario() {
      document.getElementById('book-id').value = '';
      document.getElementById('book-form').reset();
      document.getElementById('form-title').innerHTML = `<i class="fa-solid fa-plus" style="color: var(--verde-principal);"></i> Adicionar Livro`;
      document.getElementById('btn-save').innerHTML = `<i class="fa-solid fa-plus"></i> Adicionar Livro`;
      document.getElementById('btn-cancel').style.display = 'none';
    }

    function filtrarLivros() {
      const termo = document.getElementById('search-input').value.toLowerCase();
      const filtrados = listaLivros.filter(l => 
        l.titulo.toLowerCase().includes(termo) ||
        l.autor.toLowerCase().includes(termo) ||
        l.isbn.toLowerCase().includes(termo)
      );
      renderizarTabela(filtrados);
    }

    async function testarConexao() {
      if (isMock()) {
        showAlert("Modo Mock ativado.", "info");
        return;
      }
      try {
        const res = await fetch(getApiUrl(), { method: 'GET', headers: getHeaders() });
        if (res.ok) {
          showAlert("Conexão com a API estabelecida com sucesso!");
        } else {
          showAlert(`Status da resposta: ${res.status}`, "error");
        }
      } catch (e) {
        showAlert("Não foi possível conectar à API.", "error");
      }
    }
  </script>
